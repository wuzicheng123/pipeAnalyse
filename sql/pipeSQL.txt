创建数据库
CREATE DATABASE pipeanalyse;
创建用户并指定允许从任何主机连接
CREATE USER 'pipeanalyse_user'@'%' IDENTIFIED BY '123456';
GRANT ALL PRIVILEGES ON pipeanalyse.* TO 'pipeanalyse_user'@'%';
FLUSH PRIVILEGES;
新建数据库用户登录语句
mysql -u pipeanalyse_user -p
新建用户表
USE pipeAnalyse;

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    permission VARCHAR(50) DEFAULT '操作员'
);

-- 创建索引(无需执行，添加UNIQUE属性时会默认加索引)
CREATE INDEX idx_name ON users(name);

新建项目表
create table project(
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '项目ID',
    name VARCHAR(255) NOT NULL COMMENT '项目名称',
    discript VARCHAR(1000) COMMENT '项目描述',
    sampleinterval DOUBLE COMMENT '采样间距',
    wallthicknesstype VARCHAR(100) COMMENT '壁厚',
    datapath VARCHAR(500) COMMENT '数据文件路径',
    createtime TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    creator INT NOT NULL COMMENT '创建者ID，关联外键',
    PRIMARY KEY(id),
    INDEX idx_name(name),
    INDEX idx_createtime(createtime),
    INDEX idx_creator(creator)
)
